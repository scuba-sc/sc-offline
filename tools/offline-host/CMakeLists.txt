cmake_minimum_required(VERSION 3.30)
project(sc_offline_host_composition LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SCO_HOST_SOURCE_DIR "" CACHE PATH "Extracted host-source-sc kit")
if(NOT EXISTS "${SCO_HOST_SOURCE_DIR}/include/sco/game/host.h")
  message(FATAL_ERROR "Set SCO_HOST_SOURCE_DIR to a host-source-sc kit containing G1")
endif()
set(SCO_GAME_SC ON CACHE BOOL "" FORCE)
set(SCO_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${SCO_HOST_SOURCE_DIR}" sco-core EXCLUDE_FROM_ALL)
add_library(sc_offline_host_composition STATIC ../../src/offline_host.cpp)
target_link_libraries(sc_offline_host_composition PUBLIC sco_game_host)
include(CTest)
if(BUILD_TESTING)
  # The shipped gitlink predates the detailed contract. Keep its legacy fixture
  # until the SDK pin is reviewed separately; exercise reports when available.
  file(READ "${SCO_HOST_SOURCE_DIR}/include/sco/game/host.h" host_header)
  if(host_header MATCHES "class HostCapsule" AND EXISTS "${SCO_HOST_SOURCE_DIR}/src/game/host_testing.h")
    # Same production route; compile a separate host.cpp with fixed test faults,
    # as the core report fixture does. Never alter the production target.
    add_executable(host_report_test host_report_test.cpp "${SCO_HOST_SOURCE_DIR}/src/game/host.cpp")
    target_include_directories(host_report_test PRIVATE "${SCO_HOST_SOURCE_DIR}")
    target_compile_definitions(host_report_test PRIVATE SCO_GAME_HOST_REPORT_TESTING=1)
    target_link_libraries(host_report_test PRIVATE sco_game_host)
    foreach(scenario refusal reentry wrong-thread failed-start options-copy copy copy-negative)
      add_test(NAME host_report_${scenario} COMMAND host_report_test ${scenario})
      set_tests_properties(host_report_${scenario} PROPERTIES TIMEOUT 20)
    endforeach()
    # The checked-in gitlink intentionally remains on the older contract.
    file(READ "${SCO_HOST_SOURCE_DIR}/include/sco/owned_shutdown.h" shutdown_header)
    if(shutdown_header MATCHES "reap_retained_until")
      target_compile_definitions(host_report_test PRIVATE SCO_HAS_RETAINED_REAP=1)
      if(EXISTS "${SCO_HOST_SOURCE_DIR}/tests/owned_shutdown_test_access.h")
        target_compile_definitions(host_report_test PRIVATE SCO_HAS_REAP_TEST_ACCESS=1)
      endif()
      add_test(NAME host_report_retained_reap COMMAND host_report_test retained-reap)
      set_tests_properties(host_report_retained_reap PROPERTIES TIMEOUT 20)
    endif()
  endif()
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND NOT host_header MATCHES "class HostCapsule")
    add_executable(options_copy_fault_test options_copy_fault_test.cpp)
    target_link_libraries(options_copy_fault_test PRIVATE sco_game_host)
    add_test(NAME owned_capsule_options_copy_fault COMMAND options_copy_fault_test)
    set_tests_properties(owned_capsule_options_copy_fault PROPERTIES TIMEOUT 15 SKIP_RETURN_CODE 77)
  endif()
  add_executable(owned_capsule_test owned_capsule_test.cpp)
  target_link_libraries(owned_capsule_test PRIVATE sco_game_host)
  option(SCO_CAPSULE_PRIVATE_SEAM "Enable available core source-only post-start injection" ON)
  if(SCO_CAPSULE_PRIVATE_SEAM AND EXISTS "${SCO_HOST_SOURCE_DIR}/src/game/host_testing.h")
    target_include_directories(owned_capsule_test PRIVATE "${SCO_HOST_SOURCE_DIR}")
    target_compile_definitions(owned_capsule_test PRIVATE SCO_CAPSULE_PRIVATE_SEAM)
    add_test(NAME owned_capsule_after_app_start COMMAND owned_capsule_test after-app-start)
    set_tests_properties(owned_capsule_after_app_start PROPERTIES TIMEOUT 15)
  endif()
  foreach(scenario badarg admission budget host start-fault copy copy-negative)
    add_test(NAME owned_capsule_${scenario} COMMAND owned_capsule_test ${scenario})
    set_tests_properties(owned_capsule_${scenario} PROPERTIES TIMEOUT 15)
  endforeach()
  add_executable(refusal_test refusal_test.cpp ../../src/offline_host.cpp)
  target_include_directories(refusal_test PRIVATE "${SCO_HOST_SOURCE_DIR}/include")
  add_test(NAME native_activation_refusal COMMAND refusal_test)
  add_test(NAME native_activation_exception_refusal COMMAND refusal_test exception)
  add_executable(capability_refusal_test capability_refusal_test.cpp)
  target_include_directories(capability_refusal_test PRIVATE ../../src "${SCO_HOST_SOURCE_DIR}/include")
  add_test(NAME capability_refusal COMMAND capability_refusal_test)
  add_executable(registry_refusal_test registry_refusal_test.cpp)
  target_link_libraries(registry_refusal_test PRIVATE sc_offline_host_composition)
  add_test(NAME product_registry_refusal COMMAND registry_refusal_test)
  add_executable(signature_rows_test signature_rows_test.cpp)
  target_include_directories(signature_rows_test PRIVATE ../../src)
  target_link_libraries(signature_rows_test PRIVATE sco_core)
  add_test(NAME signature_rows COMMAND signature_rows_test)
endif()
