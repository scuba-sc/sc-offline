cmake_minimum_required(VERSION 3.30)
project(sc_offline_host_composition LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SCO_HOST_SOURCE_DIR "" CACHE PATH "Extracted host-source-sc kit")
if(NOT EXISTS "${SCO_HOST_SOURCE_DIR}/include/sco/game/host.h")
  message(FATAL_ERROR "Set SCO_HOST_SOURCE_DIR to a host-source-sc kit containing G1")
endif()
set(SCO_GAME_SC ON CACHE BOOL "" FORCE)
set(SCO_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${SCO_HOST_SOURCE_DIR}" sco-core EXCLUDE_FROM_ALL)
add_library(sc_offline_host_composition STATIC ../../src/offline_host.cpp)
target_link_libraries(sc_offline_host_composition PUBLIC sco_game_host)
