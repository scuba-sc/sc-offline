# Portable compile target for the supported host-source boundary; not a plugin SDK grant.
cmake_minimum_required(VERSION 3.30)
project(sc-offline-asop-shell LANGUAGES C CXX)
set(SCO_HOST_SOURCE_DIR "" CACHE PATH "Integrated Star Citizen host-source kit")
if(NOT EXISTS "${SCO_HOST_SOURCE_DIR}/include/sco/game/asop_host.h")
  message(FATAL_ERROR "A host-source dependency containing G10 is required")
endif()
option(SCO_ASOP_SELECTION_TESTS "Synthetic product startup-policy fixture" OFF)
if(SCO_ASOP_SELECTION_TESTS)
  enable_testing()
  add_executable(asop-host-selection-test ../asop_host_selection_test.cpp ../../src/asop_shell.cpp ../../src/asop_platform.cpp)
  target_include_directories(asop-host-selection-test PRIVATE "${SCO_HOST_SOURCE_DIR}/include")
  target_compile_features(asop-host-selection-test PRIVATE cxx_std_20)
  if(MSVC)
    target_compile_options(asop-host-selection-test PRIVATE /W4 /WX /permissive- /D_CRT_SECURE_NO_WARNINGS)
  endif()
  add_test(NAME asop-host-selection COMMAND asop-host-selection-test)
endif()
set(SCO_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SCO_GAME_SC ON CACHE BOOL "" FORCE)
add_subdirectory("${SCO_HOST_SOURCE_DIR}" core EXCLUDE_FROM_ALL)
add_library(sc-offline-asop-shell STATIC ../../src/asop_shell.cpp ../../src/asop_platform.cpp)
target_compile_features(sc-offline-asop-shell PUBLIC cxx_std_20)
target_include_directories(sc-offline-asop-shell PUBLIC "${SCO_HOST_SOURCE_DIR}/include")
# Static product boundary: validate its translation units without rebuilding
# unrelated game adapters from the published kit. The composing host links G10.
if(MSVC)
  target_compile_options(sc-offline-asop-shell PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(sc-offline-asop-shell PRIVATE -Wall -Wextra -Werror)
endif()
