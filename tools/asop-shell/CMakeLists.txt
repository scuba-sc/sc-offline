# Portable compile target for the supported host-source boundary; not a plugin SDK grant.
cmake_minimum_required(VERSION 3.30)
project(sc-offline-asop-shell LANGUAGES C CXX)
set(SCO_HOST_SOURCE_DIR "" CACHE PATH "Integrated Star Citizen host-source kit")
if(NOT EXISTS "${SCO_HOST_SOURCE_DIR}/include/sco/game/asop_host.h")
  message(FATAL_ERROR "A host-source dependency containing G10 is required")
endif()
set(SCO_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SCO_GAME_SC ON CACHE BOOL "" FORCE)
add_subdirectory("${SCO_HOST_SOURCE_DIR}" core EXCLUDE_FROM_ALL)
add_library(sc-offline-asop-shell STATIC ../../src/asop_shell.cpp ../../src/asop_platform.cpp)
target_compile_features(sc-offline-asop-shell PUBLIC cxx_std_20)
target_link_libraries(sc-offline-asop-shell PUBLIC sco_asop_host)
if(MSVC)
  target_compile_options(sc-offline-asop-shell PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(sc-offline-asop-shell PRIVATE -Wall -Wextra -Werror)
endif()

# Exercise process-local environment parsing and admission without native hooks.
include(CTest)
if(BUILD_TESTING)
  add_executable(asop-environment-test environment_test.cpp)
  target_compile_features(asop-environment-test PRIVATE cxx_std_20)
  target_include_directories(asop-environment-test PRIVATE "${SCO_HOST_SOURCE_DIR}/include")
  if(MSVC)
    target_compile_options(asop-environment-test PRIVATE /W4 /WX /permissive- /sdl)
  else()
    target_compile_options(asop-environment-test PRIVATE -Wall -Wextra -Werror)
  endif()
  add_test(NAME asop-environment-policy COMMAND asop-environment-test)
  add_executable(asop-start-failure-test start_failure_test.cpp ../../src/asop_shell.cpp)
  target_compile_features(asop-start-failure-test PRIVATE cxx_std_20)
  target_include_directories(asop-start-failure-test PRIVATE "${SCO_HOST_SOURCE_DIR}/include")
  if(MSVC)
    target_compile_options(asop-start-failure-test PRIVATE /W4 /WX /permissive- /sdl)
  else()
    target_compile_options(asop-start-failure-test PRIVATE -Wall -Wextra -Werror)
  endif()
  add_test(NAME asop-start-failure COMMAND asop-start-failure-test)
endif()
