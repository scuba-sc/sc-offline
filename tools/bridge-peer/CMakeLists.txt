cmake_minimum_required(VERSION 3.20)
project(sc-offline-bridge-peer LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(SCO_WIRE_INCLUDE_DIR "" CACHE PATH "Directory containing only the three published MIT wire headers")
foreach(header sc_ipc.h sc_titanlink.h sc_voxel_bridge.h)
  if(NOT EXISTS "${SCO_WIRE_INCLUDE_DIR}/${header}")
    message(FATAL_ERROR "Missing MIT wire header: ${header}; no host SDK/library required")
  endif()
endforeach()
add_executable(bridge-peer bridge_peer.cpp)
target_include_directories(bridge-peer PRIVATE "${SCO_WIRE_INCLUDE_DIR}")
target_compile_features(bridge-peer PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(bridge-peer PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(bridge-peer PRIVATE -Wall -Wextra -Werror)
endif()
# Deliberately no install rule, host target/library or default-release component.
