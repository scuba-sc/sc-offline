cmake_minimum_required(VERSION 3.20)
project(launcher_mcp_checks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(launcher_mcp_test ../launcher_mcp_test.cpp)
add_test(NAME launcher_mcp COMMAND launcher_mcp_test)
add_executable(launcher_redact_diff_test ../launcher_redact_diff_test.cpp)
add_test(NAME launcher_redact_diff COMMAND launcher_redact_diff_test)
# The dump runs the whole check suite of launcher_mcp_test.cpp and records the redaction calls it makes. That file's main() is
# renamed and given a return value in a copy written here at configure time; the original is not changed.
file(READ "${CMAKE_CURRENT_LIST_DIR}/../launcher_mcp_test.cpp" MCP_TEST_SRC)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/../launcher_mcp_test.cpp")
string(FIND "${MCP_TEST_SRC}" "int main() {" MCP_MAIN_AT)
string(FIND "${MCP_TEST_SRC}" "portable checks passed\\n\";" MCP_END_AT)
if(MCP_MAIN_AT EQUAL -1 OR MCP_END_AT EQUAL -1)
  message(FATAL_ERROR "launcher_mcp_test.cpp no longer has the main() shape that dump.cpp patches")
endif()
string(REPLACE "int main() {" "static int sco_mcp_test_main() {" MCP_TEST_SRC "${MCP_TEST_SRC}")
string(REPLACE "portable checks passed\\n\";" "portable checks passed\\n\"; return 0;" MCP_TEST_SRC "${MCP_TEST_SRC}")
get_filename_component(MCP_LAUNCHER_DIR "${CMAKE_CURRENT_LIST_DIR}/../../launcher" ABSOLUTE)
string(REPLACE "#include \"../launcher/mcp_server.h\"" "#include \"${MCP_LAUNCHER_DIR}/mcp_server.h\"" MCP_TEST_SRC "${MCP_TEST_SRC}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/launcher_mcp_test_recorded.inc" "${MCP_TEST_SRC}")
add_executable(launcher_mcp_dump dump.cpp)
target_include_directories(launcher_mcp_dump PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
# The Rust crates scol-redact and scol-mcp replay launcher-rs/tests/mcp-vectors/vectors.json. This test regenerates it from
# the C++ code and requires the committed bytes: run `launcher_mcp_dump <scenarios.json> <vectors.json>` to rewrite it.
set(MCP_VECTORS "${CMAKE_CURRENT_LIST_DIR}/../../launcher-rs/tests/mcp-vectors")
add_test(NAME launcher_mcp_vectors COMMAND launcher_mcp_dump --check "${MCP_VECTORS}/scenarios.json" "${MCP_VECTORS}/vectors.json")
if(MSVC)
  target_compile_options(launcher_mcp_dump PRIVATE /W4 /WX /utf-8)
  target_compile_options(launcher_mcp_test PRIVATE /W4 /WX)
  target_compile_options(launcher_redact_diff_test PRIVATE /W4 /WX)
else()
  target_compile_options(launcher_mcp_dump PRIVATE -Wall -Wextra -Werror)
  target_compile_options(launcher_mcp_test PRIVATE -Wall -Wextra -Werror)
  target_compile_options(launcher_redact_diff_test PRIVATE -Wall -Wextra -Werror)
endif()
