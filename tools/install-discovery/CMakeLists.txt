cmake_minimum_required(VERSION 3.20)
project(install_discovery_checks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
set(INSTALL_DISCOVERY_FIXTURE_DIR "" CACHE PATH "Synthetic Windows fixture directory")
set(INSTALL_DISCOVERY_DRIVE "" CACHE STRING "Temporary owned Windows fixture drive")
add_executable(install_discovery_test ../install_discovery_test.cpp)
add_test(NAME install_discovery COMMAND install_discovery_test)
if(MSVC)
  target_compile_options(install_discovery_test PRIVATE /W4 /WX)
else()
  target_compile_options(install_discovery_test PRIVATE -Wall -Wextra -Werror)
endif()
if(WIN32)
  add_executable(install_discovery_win32_test ../install_discovery_win32_test.cpp)
  target_link_libraries(install_discovery_win32_test PRIVATE pathcch)
  if(MSVC)
    target_compile_options(install_discovery_win32_test PRIVATE /W4 /WX)
  endif()
  add_test(NAME install_discovery_filesystem COMMAND install_discovery_win32_test "${INSTALL_DISCOVERY_FIXTURE_DIR}" "${INSTALL_DISCOVERY_DRIVE}")
endif()
