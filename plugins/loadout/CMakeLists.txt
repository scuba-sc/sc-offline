cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(sc-offline-loadout LANGUAGES CXX)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(SCO_SDK_INCLUDE_DIR "" CACHE PATH "Checksum-verified published SDK include directory")
option(SCO_LOADOUT_DEPENDENCY_TRACE "Emit MSVC dependency traces" OFF)
option(SCO_WORKFLOW_TESTS "Build synthetic public-service workflow tests" OFF)
if(NOT EXISTS "${SCO_SDK_INCLUDE_DIR}/../SHA256SUMS")
  message(FATAL_ERROR "Use an extracted published SDK checksum inventory")
endif()
foreach(header sco_api.h sc_loadout.h sc_cvars.h sc_creative.h sco_content.h sco_widgets.h)
  if(NOT EXISTS "${SCO_SDK_INCLUDE_DIR}/${header}")
    message(FATAL_ERROR "SDK public header missing: ${header}")
  endif()
endforeach()
add_library(sco-feature-loadout STATIC loadout.cpp)
target_include_directories(sco-feature-loadout PUBLIC "${SCO_SDK_INCLUDE_DIR}")
target_compile_features(sco-feature-loadout PUBLIC cxx_std_20)
set_target_properties(sco-feature-loadout PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
add_library(sc-offline-loadout MODULE external.cpp)
target_link_libraries(sc-offline-loadout PRIVATE sco-feature-loadout)
if(APPLE)
  target_link_options(sc-offline-loadout PRIVATE -Wl,-undefined,error)
elseif(NOT WIN32)
  target_link_options(sc-offline-loadout PRIVATE -Wl,--no-undefined)
endif()
set_target_properties(sc-offline-loadout PROPERTIES PREFIX "" OUTPUT_NAME "loadout" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/test-plugins/loadout/$<0:>" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/test-plugins/loadout/$<0:>")
foreach(target sco-feature-loadout sc-offline-loadout)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
    if(SCO_LOADOUT_DEPENDENCY_TRACE)
      set(trace_dir "${CMAKE_BINARY_DIR}/dependencies/${target}")
      file(MAKE_DIRECTORY "${trace_dir}")
      target_compile_options(${target} PRIVATE /sourceDependencies "${trace_dir}/")
      set(trace_props "${trace_dir}/includes.props")
      file(WRITE "${trace_props}" "<Project xmlns=\"http://schemas.microsoft.com/developer/msbuild/2003\"><Target Name=\"CrewToolchainIncludes\" BeforeTargets=\"ClCompile\"><WriteLinesToFile File=\"${trace_dir}/toolchain.txt\" Lines=\"$(IncludePath)\" Overwrite=\"true\" /></Target></Project>")
      set_target_properties(${target} PROPERTIES VS_USER_PROPS "${trace_props}")
    endif()
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  endif()
endforeach()
if(SCO_WORKFLOW_TESTS)
  enable_testing()
  add_executable(loadout-widget-test loadout_widget_test.cpp)
  add_executable(loadout-route-test loadout_route_test.cpp)
  add_executable(loadout-catalogue-test ../../tools/loadout_catalogue_test.cpp)
  target_include_directories(loadout-catalogue-test PRIVATE "${SCO_SDK_INCLUDE_DIR}")
  target_compile_features(loadout-catalogue-test PRIVATE cxx_std_20)
  target_link_libraries(loadout-catalogue-test PRIVATE ${CMAKE_DL_LIBS})
  add_dependencies(loadout-catalogue-test sc-offline-loadout)
  foreach(target loadout-widget-test loadout-route-test)
    target_link_libraries(${target} PRIVATE sco-feature-loadout)
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
    else()
      target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    endif()
  endforeach()
  add_test(NAME loadout-widget-workflow COMMAND loadout-widget-test)
  add_test(NAME loadout-public-route COMMAND loadout-route-test)
  if(MSVC)
    target_compile_options(loadout-catalogue-test PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(loadout-catalogue-test PRIVATE -Wall -Wextra -Werror)
  endif()
  add_test(NAME loadout-curated-catalogue COMMAND loadout-catalogue-test $<TARGET_FILE:sc-offline-loadout> "${CMAKE_CURRENT_SOURCE_DIR}/../../data/loadout_slots.txt")
endif()
install(TARGETS sc-offline-loadout LIBRARY DESTINATION data/plugins/loadout COMPONENT loadout RUNTIME DESTINATION data/plugins/loadout COMPONENT loadout)
install(FILES plugin.ini DESTINATION data/plugins/loadout COMPONENT loadout)
# Explicit base-list staging, not a private runtime-file read or new content format.
set(SCO_LOADOUT_BASE_LIST_DIR "" CACHE PATH "Existing product data directory to stage as a read-only loadout data pack")
if(SCO_LOADOUT_BASE_LIST_DIR)
  foreach(list items outfits buildables)
    if(NOT EXISTS "${SCO_LOADOUT_BASE_LIST_DIR}/${list}.txt")
      message(FATAL_ERROR "Loadout base list missing: ${list}.txt")
    endif()
    install(FILES "${SCO_LOADOUT_BASE_LIST_DIR}/${list}.txt" DESTINATION data/plugins/loadout_lists/lists COMPONENT loadout)
  endforeach()
  install(FILES lists-plugin.ini DESTINATION data/plugins/loadout_lists RENAME plugin.ini COMPONENT loadout)
endif()
