cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(sc-offline-loadout LANGUAGES CXX)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(SCO_SDK_INCLUDE_DIR "" CACHE PATH "Checksum-verified published SDK include directory")
option(SCO_LOADOUT_DEPENDENCY_TRACE "Emit MSVC dependency traces" OFF)
if(NOT EXISTS "${SCO_SDK_INCLUDE_DIR}/../SHA256SUMS")
  message(FATAL_ERROR "Use an extracted published SDK checksum inventory")
endif()
foreach(header sco_api.h sc_loadout.h sc_cvars.h sc_creative.h sco_content.h sco_widgets.h)
  if(NOT EXISTS "${SCO_SDK_INCLUDE_DIR}/${header}")
    message(FATAL_ERROR "SDK public header missing: ${header}")
  endif()
endforeach()
add_library(sco-feature-loadout STATIC loadout.cpp)
option(SCO_LOADOUT_ASYNC_CONSUMER "Candidate-only loadout1.2 lifecycle and logical attachments; published default unchanged" OFF)
if(SCO_LOADOUT_ASYNC_CONSUMER)
  set(SCO_LOADOUT_SDK_ARCHIVE "" CACHE FILEPATH "Exact candidate combined SDK archive")
  set(SCO_LOADOUT_SDK_SHA256 "" CACHE STRING "Independently verified candidate archive SHA-256")
  if(NOT EXISTS "${SCO_LOADOUT_SDK_ARCHIVE}")
    message(FATAL_ERROR "Async consumer requires an explicit verified SDK archive")
  endif()
  file(SHA256 "${SCO_LOADOUT_SDK_ARCHIVE}" actual_sdk_hash)
  if(NOT actual_sdk_hash STREQUAL SCO_LOADOUT_SDK_SHA256)
    message(FATAL_ERROR "Async SDK archive SHA-256 mismatch")
  endif()
  # Bind the entire public include tree, not just the archive's filename or a
  # separately supplied checksum file. Extraction is configure-time build data.
  set(archive_proof "${CMAKE_CURRENT_BINARY_DIR}/sdk-identity")
  file(ARCHIVE_EXTRACT INPUT "${SCO_LOADOUT_SDK_ARCHIVE}" DESTINATION "${archive_proof}")
  file(GLOB roots LIST_DIRECTORIES true "${archive_proof}/sco-sdk-*")
  list(LENGTH roots root_count)
  if(NOT root_count EQUAL 1)
    message(FATAL_ERROR "Expected one combined SDK archive root")
  endif()
  list(GET roots 0 archive_root)
  file(GLOB_RECURSE archive_headers CONFIGURE_DEPENDS RELATIVE "${archive_root}/include" "${archive_root}/include/*")
  file(GLOB_RECURSE selected_headers CONFIGURE_DEPENDS RELATIVE "${SCO_SDK_INCLUDE_DIR}" "${SCO_SDK_INCLUDE_DIR}/*")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${SCO_LOADOUT_SDK_ARCHIVE}")
  if(NOT archive_headers STREQUAL selected_headers)
    message(FATAL_ERROR "Selected SDK include inventory differs from verified archive")
  endif()
  foreach(header IN LISTS archive_headers)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
      "${SCO_SDK_INCLUDE_DIR}/${header}")
    file(SHA256 "${archive_root}/include/${header}" archive_header_hash)
    file(SHA256 "${SCO_SDK_INCLUDE_DIR}/${header}" selected_header_hash)
    if(NOT archive_header_hash STREQUAL selected_header_hash)
      message(FATAL_ERROR "Selected SDK header differs from verified archive: ${header}")
    endif()
  endforeach()
  target_compile_definitions(sco-feature-loadout PRIVATE SCO_LOADOUT_ASYNC_CONSUMER)
endif()
target_include_directories(sco-feature-loadout PUBLIC "${SCO_SDK_INCLUDE_DIR}")
target_compile_features(sco-feature-loadout PUBLIC cxx_std_20)
set_target_properties(sco-feature-loadout PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
add_library(sc-offline-loadout MODULE external.cpp)
target_link_libraries(sc-offline-loadout PRIVATE sco-feature-loadout)
if(APPLE)
  target_link_options(sc-offline-loadout PRIVATE -Wl,-undefined,error)
elseif(NOT WIN32)
  target_link_options(sc-offline-loadout PRIVATE -Wl,--no-undefined)
endif()
option(SCO_LOADOUT_WORKFLOW_TEST "Build actual DLL async consumer workflow fixture" OFF)
if(SCO_LOADOUT_WORKFLOW_TEST)
  if(NOT SCO_LOADOUT_ASYNC_CONSUMER)
    message(FATAL_ERROR "Async workflow requires SCO_LOADOUT_ASYNC_CONSUMER")
  endif()
  enable_testing()
  add_executable(loadout-workflow-test ../../tools/loadout_workflow_test.cpp)
  target_include_directories(loadout-workflow-test PRIVATE "${SCO_SDK_INCLUDE_DIR}")
  target_compile_features(loadout-workflow-test PRIVATE cxx_std_20)
  target_link_libraries(loadout-workflow-test PRIVATE ${CMAKE_DL_LIBS})
  add_test(NAME loadout-async-workflow COMMAND loadout-workflow-test $<TARGET_FILE:sc-offline-loadout>)
endif()
set_target_properties(sc-offline-loadout PROPERTIES PREFIX "" OUTPUT_NAME "loadout" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/test-plugins/loadout/$<0:>" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/test-plugins/loadout/$<0:>")
foreach(target sco-feature-loadout sc-offline-loadout)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
    if(SCO_LOADOUT_DEPENDENCY_TRACE)
      set(trace_dir "${CMAKE_BINARY_DIR}/dependencies/${target}")
      file(MAKE_DIRECTORY "${trace_dir}")
      target_compile_options(${target} PRIVATE /sourceDependencies "${trace_dir}/")
      set(trace_props "${trace_dir}/includes.props")
      file(WRITE "${trace_props}" "<Project xmlns=\"http://schemas.microsoft.com/developer/msbuild/2003\"><Target Name=\"CrewToolchainIncludes\" BeforeTargets=\"ClCompile\"><WriteLinesToFile File=\"${trace_dir}/toolchain.txt\" Lines=\"$(IncludePath)\" Overwrite=\"true\" /></Target></Project>")
      set_target_properties(${target} PROPERTIES VS_USER_PROPS "${trace_props}")
    endif()
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  endif()
endforeach()
install(TARGETS sc-offline-loadout LIBRARY DESTINATION data/plugins/loadout COMPONENT loadout RUNTIME DESTINATION data/plugins/loadout COMPONENT loadout)
install(FILES plugin.ini DESTINATION data/plugins/loadout COMPONENT loadout)
# Explicit base-list staging, not a private runtime-file read or new content format.
set(SCO_LOADOUT_BASE_LIST_DIR "" CACHE PATH "Existing product data directory to stage as a read-only loadout data pack")
if(SCO_LOADOUT_BASE_LIST_DIR)
  foreach(list items outfits buildables)
    if(NOT EXISTS "${SCO_LOADOUT_BASE_LIST_DIR}/${list}.txt")
      message(FATAL_ERROR "Loadout base list missing: ${list}.txt")
    endif()
    install(FILES "${SCO_LOADOUT_BASE_LIST_DIR}/${list}.txt" DESTINATION data/plugins/loadout_lists/lists COMPONENT loadout)
  endforeach()
  install(FILES lists-plugin.ini DESTINATION data/plugins/loadout_lists RENAME plugin.ini COMPONENT loadout)
endif()
