cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(sc-offline-build LANGUAGES CXX)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(SCO_SDK_INCLUDE_DIR "" CACHE PATH "Verified published SDK include directory")
if(NOT EXISTS "${SCO_SDK_INCLUDE_DIR}/../SHA256SUMS")
  message(FATAL_ERROR "Use the extracted published SDK, not host headers")
endif()
foreach(header sco_api.h sc_spawn.h sc_entities.h sc_actors.h sc_spatial.h sc_prefabs.h sc_owned.h sc_world.h sc_world_control.h)
  if(NOT EXISTS "${SCO_SDK_INCLUDE_DIR}/${header}")
    message(FATAL_ERROR "Missing published header: ${header}")
  endif()
endforeach()
add_library(sco-feature-build STATIC build.cpp)
target_include_directories(sco-feature-build PUBLIC "${SCO_SDK_INCLUDE_DIR}")
target_compile_features(sco-feature-build PUBLIC cxx_std_20)
set_target_properties(sco-feature-build PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden)
add_library(sc-offline-build MODULE external.cpp)
target_link_libraries(sc-offline-build PRIVATE sco-feature-build)
if(WIN32)
  target_link_libraries(sc-offline-build PRIVATE user32)
endif()
set_target_properties(sc-offline-build PROPERTIES PREFIX "" OUTPUT_NAME "build" CXX_VISIBILITY_PRESET hidden
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/test-plugins/build/$<0:>"
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/test-plugins/build/$<0:>")
if(APPLE)
  target_link_options(sc-offline-build PRIVATE -Wl,-undefined,error)
elseif(NOT WIN32)
  target_link_options(sc-offline-build PRIVATE -Wl,--no-undefined)
endif()
option(SCO_BUILD_DEPENDENCY_TRACE "Emit MSVC compiler dependency inventory" OFF)
foreach(target sco-feature-build sc-offline-build)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
    if(SCO_BUILD_DEPENDENCY_TRACE)
      set(trace_dir "${CMAKE_BINARY_DIR}/dependencies/${target}")
      file(MAKE_DIRECTORY "${trace_dir}")
      target_compile_options(${target} PRIVATE /sourceDependencies "${trace_dir}/")
      set(trace_props "${trace_dir}/includes.props")
      file(WRITE "${trace_props}" "<Project xmlns=\"http://schemas.microsoft.com/developer/msbuild/2003\"><Target Name=\"WorkflowToolchainIncludes\" BeforeTargets=\"ClCompile\"><WriteLinesToFile File=\"${trace_dir}/toolchain.txt\" Lines=\"$(IncludePath)\" Overwrite=\"true\" /></Target></Project>")
      set_target_properties(${target} PROPERTIES VS_USER_PROPS "${trace_props}")
    endif()
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  endif()
endforeach()
configure_file(plugin.ini "${CMAKE_BINARY_DIR}/test-plugins/build/plugin.ini" COPYONLY)
install(TARGETS sc-offline-build LIBRARY DESTINATION data/plugins/build COMPONENT build
  RUNTIME DESTINATION data/plugins/build COMPONENT build)
install(FILES plugin.ini DESTINATION data/plugins/build COMPONENT build)
option(SCO_WORKFLOW_TESTS "Build dynamic build workflow" OFF)
if(SCO_WORKFLOW_TESTS)
  enable_testing()
  add_executable(build-workflow-test ../../tools/build_workflow_test.cpp)
  target_compile_features(build-workflow-test PRIVATE cxx_std_20)
  target_include_directories(build-workflow-test PRIVATE "${SCO_SDK_INCLUDE_DIR}")
  target_link_libraries(build-workflow-test PRIVATE ${CMAKE_DL_LIBS})
  add_dependencies(build-workflow-test sc-offline-build)
  if(MSVC)
    target_compile_options(build-workflow-test PRIVATE /W4 /WX /permissive-)
  endif()
  add_test(NAME build-workflow COMMAND build-workflow-test $<TARGET_FILE:sc-offline-build>)
  set_tests_properties(build-workflow PROPERTIES TIMEOUT 30)
endif()
