cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(sc-offline-crew LANGUAGES CXX)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(SCO_SDK_INCLUDE_DIR "" CACHE PATH "Checksum-verified published SDK include directory")
option(SCO_WORKFLOW_TESTS "Build dynamic synthetic crew fixture" OFF)
option(SCO_CREW_DEPENDENCY_TRACE "Emit MSVC dependency traces" OFF)
if(NOT EXISTS "${SCO_SDK_INCLUDE_DIR}/../SHA256SUMS")
  message(FATAL_ERROR "Use an extracted published SDK checksum inventory")
endif()
foreach(header sco_widgets.h sco_content.h sco_api.h sc_actors.h sc_spatial.h sc_vehicles.h)
  if(NOT EXISTS "${SCO_SDK_INCLUDE_DIR}/${header}")
    message(FATAL_ERROR "SDK public header missing: ${header}")
  endif()
endforeach()
add_library(sco-feature-crew STATIC crew.cpp)
target_include_directories(sco-feature-crew PUBLIC "${SCO_SDK_INCLUDE_DIR}")
target_compile_features(sco-feature-crew PUBLIC cxx_std_20)
set_target_properties(sco-feature-crew PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
add_library(sc-offline-crew MODULE external.cpp)
target_link_libraries(sc-offline-crew PRIVATE sco-feature-crew)
if(APPLE)
  target_link_options(sc-offline-crew PRIVATE -Wl,-undefined,error)
elseif(NOT WIN32)
  target_link_options(sc-offline-crew PRIVATE -Wl,--no-undefined)
endif()
set_target_properties(sc-offline-crew PROPERTIES PREFIX "" OUTPUT_NAME "crew" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/test-plugins/crew/$<0:>" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/test-plugins/crew/$<0:>")
foreach(target sco-feature-crew sc-offline-crew)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
    if(SCO_CREW_DEPENDENCY_TRACE)
      set(trace_dir "${CMAKE_BINARY_DIR}/dependencies/${target}")
      file(MAKE_DIRECTORY "${trace_dir}")
      target_compile_options(${target} PRIVATE /sourceDependencies "${trace_dir}/")
      set(trace_props "${trace_dir}/includes.props")
      file(WRITE "${trace_props}" "<Project xmlns=\"http://schemas.microsoft.com/developer/msbuild/2003\"><Target Name=\"CrewToolchainIncludes\" BeforeTargets=\"ClCompile\"><WriteLinesToFile File=\"${trace_dir}/toolchain.txt\" Lines=\"$(IncludePath)\" Overwrite=\"true\" /></Target></Project>")
      set_target_properties(${target} PROPERTIES VS_USER_PROPS "${trace_props}")
    endif()
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  endif()
endforeach()
install(TARGETS sc-offline-crew LIBRARY DESTINATION data/plugins/crew COMPONENT crew RUNTIME DESTINATION data/plugins/crew COMPONENT crew)
install(FILES plugin.ini DESTINATION data/plugins/crew COMPONENT crew)
if(SCO_WORKFLOW_TESTS)
  enable_testing()
  add_executable(crew-module-test ../../tools/crew_workflow_test.cpp)
  target_include_directories(crew-module-test PRIVATE "${SCO_SDK_INCLUDE_DIR}")
  target_compile_features(crew-module-test PRIVATE cxx_std_20)
  target_link_libraries(crew-module-test PRIVATE ${CMAKE_DL_LIBS})
  add_dependencies(crew-module-test sc-offline-crew)
  add_test(NAME crew-module-workflow COMMAND crew-module-test $<TARGET_FILE:sc-offline-crew>)
endif()
