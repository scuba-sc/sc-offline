cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(sc-offline-creative LANGUAGES CXX)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(SCO_SDK_INCLUDE_DIR "" CACHE PATH "Verified published SDK public includes")
option(SCO_CREATIVE_DEPENDENCY_TRACE "Emit MSVC creative compiler dependencies" OFF)
option(SCO_CREATIVE_SANITIZERS "Instrument the development module and focused test" OFF)
option(SCO_WORKFLOW_TESTS "Build synthetic workflow tests" OFF)
if(NOT EXISTS "${SCO_SDK_INCLUDE_DIR}/../SHA256SUMS")
  message(FATAL_ERROR "Use an extracted published SDK with its checksum inventory")
endif()
foreach(header sco_api.h sc_creative.h sco_settings.h)
  if(NOT EXISTS "${SCO_SDK_INCLUDE_DIR}/${header}")
    message(FATAL_ERROR "SDK public header missing: ${header}")
  endif()
endforeach()
add_library(sco-feature-creative STATIC creative.cpp)
target_include_directories(sco-feature-creative PUBLIC "${SCO_SDK_INCLUDE_DIR}")
target_compile_features(sco-feature-creative PUBLIC cxx_std_20)
set_target_properties(sco-feature-creative PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(sc-offline-creative MODULE external.cpp)
target_link_libraries(sc-offline-creative PRIVATE sco-feature-creative)
if(APPLE)
  target_link_options(sc-offline-creative PRIVATE -Wl,-undefined,error)
elseif(NOT WIN32)
  target_link_options(sc-offline-creative PRIVATE -Wl,--no-undefined)
endif()
set_target_properties(sc-offline-creative PROPERTIES PREFIX "" OUTPUT_NAME "creative"
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/test-plugins/creative/$<0:>"
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/test-plugins/creative/$<0:>")
foreach(target sco-feature-creative sc-offline-creative)
  set_target_properties(${target} PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
    if(SCO_CREATIVE_DEPENDENCY_TRACE)
      set(trace_dir "${CMAKE_BINARY_DIR}/dependencies/${target}")
      file(MAKE_DIRECTORY "${trace_dir}")
      target_compile_options(${target} PRIVATE /sourceDependencies "${trace_dir}/")
      set(props "${trace_dir}/includes.props")
      file(WRITE "${props}"
        "<Project xmlns=\"http://schemas.microsoft.com/developer/msbuild/2003\"><Target Name=\"CreativeToolchainIncludes\" BeforeTargets=\"ClCompile\"><WriteLinesToFile File=\"${trace_dir}/toolchain.txt\" Lines=\"$(IncludePath)\" Overwrite=\"true\" /></Target></Project>")
      set_target_properties(${target} PROPERTIES VS_USER_PROPS "${props}")
    endif()
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  endif()
endforeach()
configure_file(plugin.ini "${CMAKE_BINARY_DIR}/test-plugins/creative/plugin.ini" COPYONLY)
configure_file(disabled "${CMAKE_BINARY_DIR}/test-plugins/creative/disabled" COPYONLY)
install(TARGETS sc-offline-creative LIBRARY DESTINATION data/plugins/creative COMPONENT creative
  RUNTIME DESTINATION data/plugins/creative COMPONENT creative)
install(FILES plugin.ini disabled DESTINATION data/plugins/creative COMPONENT creative)
if(SCO_WORKFLOW_TESTS)
  enable_testing()
  add_executable(creative-module-test ../../tools/creative_module_test.cpp)
  target_compile_features(creative-module-test PRIVATE cxx_std_20)
  target_include_directories(creative-module-test PRIVATE "${SCO_SDK_INCLUDE_DIR}")
  target_link_libraries(creative-module-test PRIVATE ${CMAKE_DL_LIBS})
  add_dependencies(creative-module-test sc-offline-creative)
  if(MSVC)
    target_compile_options(creative-module-test PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(creative-module-test PRIVATE -Wall -Wextra -Werror)
  endif()
  add_test(NAME creative-module-workflow COMMAND creative-module-test $<TARGET_FILE:sc-offline-creative>)
endif()
if(SCO_CREATIVE_SANITIZERS)
  if(MSVC OR NOT SCO_WORKFLOW_TESTS)
    message(FATAL_ERROR "Sanitizer proof requires POSIX development workflow tests")
  endif()
  foreach(target sco-feature-creative sc-offline-creative creative-module-test)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  endforeach()
  foreach(target sc-offline-creative creative-module-test)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endforeach()
  # Pinned SDK callback registry intentionally lives for the process lifetime.
  set_tests_properties(creative-module-workflow PROPERTIES ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")
endif()
