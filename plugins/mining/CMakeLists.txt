cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(sc-offline-mining LANGUAGES CXX)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(SCO_SDK_INCLUDE_DIR "" CACHE PATH "Extracted public core + game SDK")
option(SCO_WORKFLOW_TESTS "Build focused synthetic mining workflow" OFF)
if(NOT EXISTS "${SCO_SDK_INCLUDE_DIR}/../SHA256SUMS")
  message(FATAL_ERROR "Use an extracted SDK archive, never private host headers")
endif()
foreach(header sco_api.h sco_settings.h sc_mining.h)
  if(NOT EXISTS "${SCO_SDK_INCLUDE_DIR}/${header}")
    message(FATAL_ERROR "Required SDK header missing: ${header}")
  endif()
endforeach()
add_library(sc-offline-mining MODULE mining.cpp)
target_include_directories(sc-offline-mining PRIVATE "${SCO_SDK_INCLUDE_DIR}")
target_compile_features(sc-offline-mining PRIVATE cxx_std_20)
set_target_properties(sc-offline-mining PROPERTIES PREFIX "" OUTPUT_NAME "mining"
  CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins/mining/$<0:>"
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins/mining/$<0:>")
if(MSVC)
  target_compile_options(sc-offline-mining PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(sc-offline-mining PRIVATE -Wall -Wextra -Werror)
  if(APPLE)
    target_link_options(sc-offline-mining PRIVATE -Wl,-undefined,error)
  else()
    target_link_options(sc-offline-mining PRIVATE -Wl,--no-undefined)
  endif()
endif()
configure_file(plugin.ini "${CMAKE_BINARY_DIR}/plugins/mining/plugin.ini" COPYONLY)
install(TARGETS sc-offline-mining LIBRARY DESTINATION data/plugins/mining COMPONENT mining
  RUNTIME DESTINATION data/plugins/mining COMPONENT mining)
install(FILES plugin.ini import_settings.py README.md DESTINATION data/plugins/mining COMPONENT mining)
if(SCO_WORKFLOW_TESTS)
  enable_testing()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(mining-module-test ../../tools/mining_test.cpp)
  target_compile_features(mining-module-test PRIVATE cxx_std_20)
  target_include_directories(mining-module-test PRIVATE "${SCO_SDK_INCLUDE_DIR}")
  target_link_libraries(mining-module-test PRIVATE ${CMAKE_DL_LIBS})
  add_dependencies(mining-module-test sc-offline-mining)
  if(MSVC)
    target_compile_options(mining-module-test PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(mining-module-test PRIVATE -Wall -Wextra -Werror)
  endif()
  add_test(NAME mining-workflow COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_workflow.py"
    "$<TARGET_FILE:mining-module-test>" "$<TARGET_FILE:sc-offline-mining>")
endif()
