cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(sc-offline-contracts LANGUAGES CXX)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(SCO_SDK_INCLUDE_DIR "" CACHE PATH "Verified published SDK include directory")
if(NOT EXISTS "${SCO_SDK_INCLUDE_DIR}/../SHA256SUMS")
  message(FATAL_ERROR "Use the extracted published SDK, not host headers")
endif()
foreach(header sco_api.h sc_missions.h sc_economy.h sco_content.h)
  if(NOT EXISTS "${SCO_SDK_INCLUDE_DIR}/${header}")
    message(FATAL_ERROR "Missing published header: ${header}")
  endif()
endforeach()
add_library(sco-feature-contracts STATIC contracts.cpp)
target_include_directories(sco-feature-contracts PUBLIC "${SCO_SDK_INCLUDE_DIR}")
target_compile_features(sco-feature-contracts PUBLIC cxx_std_20)
set_target_properties(sco-feature-contracts PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden)
add_library(sc-offline-contracts MODULE external.cpp)
target_link_libraries(sc-offline-contracts PRIVATE sco-feature-contracts)
set_target_properties(sc-offline-contracts PROPERTIES PREFIX "" OUTPUT_NAME "contracts" CXX_VISIBILITY_PRESET hidden
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/test-plugins/contracts/$<0:>"
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/test-plugins/contracts/$<0:>")
if(APPLE)
  target_link_options(sc-offline-contracts PRIVATE -Wl,-undefined,error)
elseif(NOT WIN32)
  target_link_options(sc-offline-contracts PRIVATE -Wl,--no-undefined)
endif()
option(SCO_CONTRACTS_DEPENDENCY_TRACE "Emit MSVC compiler dependency inventory" OFF)
foreach(target sco-feature-contracts sc-offline-contracts)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
    if(SCO_CONTRACTS_DEPENDENCY_TRACE)
      set(trace_dir "${CMAKE_BINARY_DIR}/dependencies/${target}")
      file(MAKE_DIRECTORY "${trace_dir}")
      target_compile_options(${target} PRIVATE /sourceDependencies "${trace_dir}/")
      set(trace_props "${trace_dir}/includes.props")
      file(WRITE "${trace_props}" "<Project xmlns=\"http://schemas.microsoft.com/developer/msbuild/2003\"><Target Name=\"WorkflowToolchainIncludes\" BeforeTargets=\"ClCompile\"><WriteLinesToFile File=\"${trace_dir}/toolchain.txt\" Lines=\"$(IncludePath)\" Overwrite=\"true\" /></Target></Project>")
      set_target_properties(${target} PROPERTIES VS_USER_PROPS "${trace_props}")
    endif()
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  endif()
endforeach()
configure_file(plugin.ini "${CMAKE_BINARY_DIR}/test-plugins/contracts/plugin.ini" COPYONLY)
install(TARGETS sc-offline-contracts LIBRARY DESTINATION data/plugins/contracts COMPONENT contracts
  RUNTIME DESTINATION data/plugins/contracts COMPONENT contracts)
install(FILES plugin.ini DESTINATION data/plugins/contracts COMPONENT contracts)
option(SCO_WORKFLOW_TESTS "Build the synthetic dynamic module workflow" OFF)
if(SCO_WORKFLOW_TESTS)
  enable_testing()
  add_executable(contracts-module-test ../../tools/contracts_workflow_test.cpp)
  target_compile_features(contracts-module-test PRIVATE cxx_std_20)
  target_include_directories(contracts-module-test PRIVATE "${SCO_SDK_INCLUDE_DIR}")
  target_link_libraries(contracts-module-test PRIVATE ${CMAKE_DL_LIBS})
  add_dependencies(contracts-module-test sc-offline-contracts)
  foreach(slice read accept progress script)
    add_test(NAME contracts-${slice} COMMAND contracts-module-test $<TARGET_FILE:sc-offline-contracts> ${slice})
  endforeach()
endif()
