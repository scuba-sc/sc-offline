# Independent Discord consumer, also addable from the standalone plugins build.
cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(sc-offline-discord LANGUAGES CXX)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  enable_testing()
endif()
set(SCO_SDK_INCLUDE_DIR "" CACHE PATH "Extracted published SDK include directory")
option(SCO_DISCORD_DEPENDENCY_TRACE "Emit MSVC Discord compiler dependencies" OFF)
option(SCO_WORKFLOW_TESTS "Build targeted synthetic module tests" OFF)
if(NOT EXISTS "${SCO_SDK_INCLUDE_DIR}/../SHA256SUMS")
  message(FATAL_ERROR "Use an extracted published SDK, not host headers")
endif()
foreach(header sco_api.h sco_settings.h)
  if(NOT EXISTS "${SCO_SDK_INCLUDE_DIR}/${header}")
    message(FATAL_ERROR "SDK public header missing: ${header}")
  endif()
endforeach()
add_library(sco-feature-discord STATIC discord.cpp)
target_compile_definitions(sco-feature-discord PRIVATE DISCORD_EXTERNAL_ADAPTER)
target_include_directories(sco-feature-discord PUBLIC "${SCO_SDK_INCLUDE_DIR}")
target_compile_features(sco-feature-discord PUBLIC cxx_std_20)
set_target_properties(sco-feature-discord PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(sc-offline-discord MODULE external.cpp)
target_link_libraries(sc-offline-discord PRIVATE sco-feature-discord)
if(APPLE)
  target_link_options(sc-offline-discord PRIVATE -Wl,-undefined,error)
elseif(NOT WIN32)
  target_link_options(sc-offline-discord PRIVATE -Wl,--no-undefined)
endif()
foreach(target sco-feature-discord sc-offline-discord)
  set_target_properties(${target} PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
    if(SCO_DISCORD_DEPENDENCY_TRACE)
      set(trace_dir "${CMAKE_BINARY_DIR}/dependencies/${target}")
      file(MAKE_DIRECTORY "${trace_dir}")
      target_compile_options(${target} PRIVATE /sourceDependencies "${trace_dir}/")
      set(trace_props "${trace_dir}/includes.props")
      file(WRITE "${trace_props}"
        "<Project xmlns=\"http://schemas.microsoft.com/developer/msbuild/2003\"><Target Name=\"WorkflowToolchainIncludes\" BeforeTargets=\"ClCompile\"><WriteLinesToFile File=\"${trace_dir}/toolchain.txt\" Lines=\"$(IncludePath)\" Overwrite=\"true\" /></Target></Project>")
      set_target_properties(${target} PROPERTIES VS_USER_PROPS "${trace_props}")
    endif()
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  endif()
endforeach()
set_target_properties(sc-offline-discord PROPERTIES PREFIX "" OUTPUT_NAME discord
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/test-plugins/discord/$<0:>"
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/test-plugins/discord/$<0:>")
configure_file(plugin.ini "${CMAKE_BINARY_DIR}/test-plugins/discord/plugin.ini" COPYONLY)
configure_file(disabled "${CMAKE_BINARY_DIR}/test-plugins/discord/disabled" COPYONLY)
install(TARGETS sc-offline-discord LIBRARY DESTINATION data/plugins/discord COMPONENT discord
  RUNTIME DESTINATION data/plugins/discord COMPONENT discord)
install(FILES plugin.ini disabled DESTINATION data/plugins/discord COMPONENT discord)
if(SCO_WORKFLOW_TESTS)
  add_executable(discord-module-test ../../tools/discord_test.cpp)
  target_compile_features(discord-module-test PRIVATE cxx_std_20)
  target_include_directories(discord-module-test PRIVATE "${SCO_SDK_INCLUDE_DIR}")
  target_link_libraries(discord-module-test PRIVATE ${CMAKE_DL_LIBS})
  add_dependencies(discord-module-test sc-offline-discord)
  add_test(NAME discord-module-workflow COMMAND discord-module-test $<TARGET_FILE:sc-offline-discord>)
endif()
