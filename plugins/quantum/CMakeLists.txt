cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(sc-offline-quantum LANGUAGES CXX)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(SCO_SDK_INCLUDE_DIR "" CACHE PATH "Extracted public SDK include directory")
if(NOT EXISTS "${SCO_SDK_INCLUDE_DIR}/../SHA256SUMS")
  message(FATAL_ERROR "Use an extracted SDK archive, never private host headers")
endif()
foreach(header sco_api.h sc_spatial.h sc_game_spatial.h sc_relocation.h sc_quantum.h sco_storage.h sco_settings.h sco_ui.h sco_widgets.h sco_content.h)
  if(NOT EXISTS "${SCO_SDK_INCLUDE_DIR}/${header}")
    message(FATAL_ERROR "Required SDK header missing: ${header}")
  endif()
endforeach()
add_library(sco-feature-quantum STATIC quantum.cpp)
target_include_directories(sco-feature-quantum PUBLIC "${SCO_SDK_INCLUDE_DIR}")
target_compile_features(sco-feature-quantum PUBLIC cxx_std_20)
set_target_properties(sco-feature-quantum PROPERTIES POSITION_INDEPENDENT_CODE ON
  CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
if(WIN32)
  target_link_libraries(sco-feature-quantum PUBLIC user32)
endif()
add_library(sc-offline-quantum MODULE external.cpp)
target_link_libraries(sc-offline-quantum PRIVATE sco-feature-quantum)
set_target_properties(sc-offline-quantum PROPERTIES PREFIX "" OUTPUT_NAME "quantum"
  CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins/quantum/$<0:>"
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins/quantum/$<0:>")
if(APPLE)
  target_link_options(sc-offline-quantum PRIVATE -Wl,-undefined,error)
elseif(NOT WIN32)
  target_link_options(sc-offline-quantum PRIVATE -Wl,--no-undefined)
endif()
option(SCO_QUANTUM_DEPENDENCY_TRACE "Emit MSVC compiler dependency inventory" OFF)
foreach(target sco-feature-quantum sc-offline-quantum)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
    if(SCO_QUANTUM_DEPENDENCY_TRACE)
      set(trace_dir "${CMAKE_BINARY_DIR}/dependencies/${target}")
      file(MAKE_DIRECTORY "${trace_dir}")
      target_compile_options(${target} PRIVATE /sourceDependencies "${trace_dir}/")
      set(trace_props "${trace_dir}/includes.props")
      file(WRITE "${trace_props}" "<Project xmlns=\"http://schemas.microsoft.com/developer/msbuild/2003\"><Target Name=\"WorkflowToolchainIncludes\" BeforeTargets=\"ClCompile\"><WriteLinesToFile File=\"${trace_dir}/toolchain.txt\" Lines=\"$(IncludePath)\" Overwrite=\"true\" /></Target></Project>")
      set_target_properties(${target} PROPERTIES VS_USER_PROPS "${trace_props}")
    endif()
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  endif()
endforeach()
option(SCO_QUANTUM_TEST "Build one synthetic actual-module quantum test" OFF)
if(SCO_QUANTUM_TEST)
  enable_testing()
  add_executable(quantum-workflow-test ../../tools/quantum_workflow_test.cpp)
  target_include_directories(quantum-workflow-test PRIVATE "${SCO_SDK_INCLUDE_DIR}")
  target_compile_features(quantum-workflow-test PRIVATE cxx_std_20)
  target_link_libraries(quantum-workflow-test PRIVATE ${CMAKE_DL_LIBS})
  add_dependencies(quantum-workflow-test sc-offline-quantum)
  add_test(NAME quantum-workflow COMMAND quantum-workflow-test $<TARGET_FILE:sc-offline-quantum>)
  add_executable(quantum-content-test ../../tools/quantum_content_test.cpp)
  target_include_directories(quantum-content-test PRIVATE "${SCO_SDK_INCLUDE_DIR}")
  target_compile_features(quantum-content-test PRIVATE cxx_std_20)
  target_link_libraries(quantum-content-test PRIVATE ${CMAKE_DL_LIBS})
  add_dependencies(quantum-content-test sc-offline-quantum)
  add_test(NAME quantum-content COMMAND quantum-content-test $<TARGET_FILE:sc-offline-quantum>)
endif()
configure_file(plugin.ini "${CMAKE_BINARY_DIR}/plugins/quantum/plugin.ini" COPYONLY)
install(TARGETS sc-offline-quantum LIBRARY DESTINATION data/plugins/quantum COMPONENT quantum
  RUNTIME DESTINATION data/plugins/quantum COMPONENT quantum)
install(FILES plugin.ini DESTINATION data/plugins/quantum COMPONENT quantum)
