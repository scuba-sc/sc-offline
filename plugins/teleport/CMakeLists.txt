cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(sc-offline-teleport LANGUAGES CXX)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(SCO_SDK_INCLUDE_DIR "" CACHE PATH "Extracted public SDK include directory")
if(NOT EXISTS "${SCO_SDK_INCLUDE_DIR}/../SHA256SUMS")
  message(FATAL_ERROR "Use an extracted SDK archive, never private host headers")
endif()
foreach(header sco_api.h sc_spatial.h sc_game_spatial.h sco_storage.h sco_settings.h sco_ui.h)
  if(NOT EXISTS "${SCO_SDK_INCLUDE_DIR}/${header}")
    message(FATAL_ERROR "Required SDK header missing: ${header}")
  endif()
endforeach()
add_library(sc-offline-teleport MODULE teleport.cpp)
target_include_directories(sc-offline-teleport PRIVATE "${SCO_SDK_INCLUDE_DIR}")
target_compile_features(sc-offline-teleport PRIVATE cxx_std_20)
set_target_properties(sc-offline-teleport PROPERTIES PREFIX "" OUTPUT_NAME "teleport"
  CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins/teleport/$<0:>"
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins/teleport/$<0:>")
if(MSVC)
  target_compile_options(sc-offline-teleport PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(sc-offline-teleport PRIVATE -Wall -Wextra -Werror)
  if(APPLE)
    target_link_options(sc-offline-teleport PRIVATE -Wl,-undefined,error)
  else()
    target_link_options(sc-offline-teleport PRIVATE -Wl,--no-undefined)
  endif()
endif()
configure_file(plugin.ini "${CMAKE_BINARY_DIR}/plugins/teleport/plugin.ini" COPYONLY)
install(TARGETS sc-offline-teleport LIBRARY DESTINATION data/plugins/teleport COMPONENT teleport
  RUNTIME DESTINATION data/plugins/teleport COMPONENT teleport)
install(FILES plugin.ini DESTINATION data/plugins/teleport COMPONENT teleport)
