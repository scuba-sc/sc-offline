cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(sc-offline-teleport LANGUAGES CXX)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(SCO_SDK_INCLUDE_DIR "" CACHE PATH "Extracted public SDK include directory")
if(NOT EXISTS "${SCO_SDK_INCLUDE_DIR}/../SHA256SUMS")
  message(FATAL_ERROR "Use an extracted SDK archive, never private host headers")
endif()
foreach(header sco_api.h sc_spatial.h sc_game_spatial.h sc_relocation.h sco_storage.h sco_settings.h sco_ui.h)
  if(NOT EXISTS "${SCO_SDK_INCLUDE_DIR}/${header}")
    message(FATAL_ERROR "Required SDK header missing: ${header}")
  endif()
endforeach()
add_library(sco-feature-teleport STATIC teleport.cpp)
target_include_directories(sco-feature-teleport PUBLIC "${SCO_SDK_INCLUDE_DIR}")
target_compile_features(sco-feature-teleport PUBLIC cxx_std_20)
set_target_properties(sco-feature-teleport PROPERTIES POSITION_INDEPENDENT_CODE ON
  CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
add_library(sc-offline-teleport MODULE external.cpp)
target_link_libraries(sc-offline-teleport PRIVATE sco-feature-teleport)
set_target_properties(sc-offline-teleport PROPERTIES PREFIX "" OUTPUT_NAME "teleport"
  CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins/teleport/$<0:>"
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins/teleport/$<0:>")
if(APPLE)
  target_link_options(sc-offline-teleport PRIVATE -Wl,-undefined,error)
elseif(NOT WIN32)
  target_link_options(sc-offline-teleport PRIVATE -Wl,--no-undefined)
endif()
option(SCO_TELEPORT_DEPENDENCY_TRACE "Emit MSVC compiler dependency inventory" OFF)
foreach(target sco-feature-teleport sc-offline-teleport)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
    if(SCO_TELEPORT_DEPENDENCY_TRACE)
      set(trace_dir "${CMAKE_BINARY_DIR}/dependencies/${target}")
      file(MAKE_DIRECTORY "${trace_dir}")
      target_compile_options(${target} PRIVATE /sourceDependencies "${trace_dir}/")
      set(trace_props "${trace_dir}/includes.props")
      file(WRITE "${trace_props}" "<Project xmlns=\"http://schemas.microsoft.com/developer/msbuild/2003\"><Target Name=\"WorkflowToolchainIncludes\" BeforeTargets=\"ClCompile\"><WriteLinesToFile File=\"${trace_dir}/toolchain.txt\" Lines=\"$(IncludePath)\" Overwrite=\"true\" /></Target></Project>")
      set_target_properties(${target} PROPERTIES VS_USER_PROPS "${trace_props}")
    endif()
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  endif()
endforeach()
option(SCO_TRAVEL_TEST "Build the synthetic travel model test" OFF)
if(SCO_TRAVEL_TEST)
  enable_testing()
  add_executable(travel-model-test ../../tools/travel_workflow_test.cpp)
  target_include_directories(travel-model-test PRIVATE "${SCO_SDK_INCLUDE_DIR}")
  target_compile_features(travel-model-test PRIVATE cxx_std_20)
  add_test(NAME travel-model COMMAND travel-model-test)
endif()
configure_file(plugin.ini "${CMAKE_BINARY_DIR}/plugins/teleport/plugin.ini" COPYONLY)
install(TARGETS sc-offline-teleport LIBRARY DESTINATION data/plugins/teleport COMPONENT teleport
  RUNTIME DESTINATION data/plugins/teleport COMPONENT teleport)
install(FILES plugin.ini DESTINATION data/plugins/teleport COMPONENT teleport)
option(SCO_WORKFLOW_TESTS "Build actual relocation consumer workflow" OFF)
if(SCO_WORKFLOW_TESTS)
  enable_testing()
  add_executable(relocation-workflow-test ../../tools/relocation_workflow_test.cpp)
  target_compile_features(relocation-workflow-test PRIVATE cxx_std_20)
  target_include_directories(relocation-workflow-test PRIVATE "${SCO_SDK_INCLUDE_DIR}")
  target_link_libraries(relocation-workflow-test PRIVATE ${CMAKE_DL_LIBS})
  add_dependencies(relocation-workflow-test sc-offline-teleport)
  if(MSVC)
    target_compile_options(relocation-workflow-test PRIVATE /W4 /WX /permissive-)
  endif()
  add_test(NAME relocation-workflow COMMAND relocation-workflow-test $<TARGET_FILE:sc-offline-teleport>)
  set_tests_properties(relocation-workflow PROPERTIES TIMEOUT 30)
endif()
