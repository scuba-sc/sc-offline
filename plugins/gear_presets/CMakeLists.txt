cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(sc-offline-gear-presets LANGUAGES CXX)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(SCO_SDK_INCLUDE_DIR "" CACHE PATH "Checksum-verified extracted SDK include directory")
option(SCO_GEAR_PRESETS_TESTS "Build the bounded synthetic model test" OFF)
set(SCO_CORE_HOST_TEST_ROOT "" CACHE PATH "Optional matching core source for host-only test fixtures")
set(SCO_CORE_HOST_BUILD_DIR "" CACHE PATH "Optional matching core build for host-only test fixtures")
if(NOT EXISTS "${SCO_SDK_INCLUDE_DIR}/../SHA256SUMS")
  message(FATAL_ERROR "Use an extracted SDK with checksum inventory")
endif()
foreach(header sco_api.h sco_storage.h sco_widgets.h sc_inventory.h scosdk/storage.hpp scosdk/game/inventory_layout.hpp)
  if(NOT EXISTS "${SCO_SDK_INCLUDE_DIR}/${header}")
    message(FATAL_ERROR "SDK header missing: ${header}")
  endif()
endforeach()
add_library(sc-offline-gear-presets MODULE plugin.cpp model.cpp store.cpp)
target_include_directories(sc-offline-gear-presets PRIVATE "${SCO_SDK_INCLUDE_DIR}")
target_compile_features(sc-offline-gear-presets PRIVATE cxx_std_20)
set_target_properties(sc-offline-gear-presets PROPERTIES PREFIX "" OUTPUT_NAME "gear_presets" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
if(APPLE)
  target_link_options(sc-offline-gear-presets PRIVATE -Wl,-undefined,error)
elseif(NOT WIN32)
  target_link_options(sc-offline-gear-presets PRIVATE -Wl,--no-undefined)
endif()
if(MSVC)
  target_compile_options(sc-offline-gear-presets PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(sc-offline-gear-presets PRIVATE -Wall -Wextra -Werror)
endif()
if(SCO_GEAR_PRESETS_TESTS)
  enable_testing()
  add_executable(gear-presets-model-test model.cpp model_test.cpp)
  target_include_directories(gear-presets-model-test PRIVATE "${SCO_SDK_INCLUDE_DIR}")
  target_compile_features(gear-presets-model-test PRIVATE cxx_std_20)
  if(MSVC)
    target_compile_options(gear-presets-model-test PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(gear-presets-model-test PRIVATE -Wall -Wextra -Werror)
  endif()
  add_test(NAME gear-presets-model COMMAND gear-presets-model-test)
  if(SCO_CORE_HOST_TEST_ROOT AND SCO_CORE_HOST_BUILD_DIR)
    add_executable(gear-presets-storage-test model.cpp store.cpp ../../tools/gear_presets_storage_test.cpp)
    add_executable(gear-presets-workflow-test ../../tools/gear_presets_workflow_test.cpp
      "${SCO_CORE_HOST_TEST_ROOT}/src/game/services/inventory.cpp"
      "${SCO_CORE_HOST_TEST_ROOT}/src/game/services/inventory_layout.cpp")
    foreach(target gear-presets-storage-test gear-presets-workflow-test)
      target_include_directories(${target} PRIVATE "${SCO_SDK_INCLUDE_DIR}" "${SCO_CORE_HOST_TEST_ROOT}/include" "${SCO_CORE_HOST_TEST_ROOT}/src" "${CMAKE_CURRENT_SOURCE_DIR}")
      target_compile_features(${target} PRIVATE cxx_std_20)
      if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
        target_link_libraries(${target} PRIVATE "${SCO_CORE_HOST_BUILD_DIR}/sco_widgets.lib" "${SCO_CORE_HOST_BUILD_DIR}/sco_storage.lib"
          "${SCO_CORE_HOST_BUILD_DIR}/sco_host.lib" "${SCO_CORE_HOST_BUILD_DIR}/sco_runtime.lib"
          "${SCO_CORE_HOST_BUILD_DIR}/sco_core.lib" "${SCO_CORE_HOST_BUILD_DIR}/sco_sqlite.lib")
      else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
        target_link_libraries(${target} PRIVATE "${SCO_CORE_HOST_BUILD_DIR}/libsco_widgets.a" "${SCO_CORE_HOST_BUILD_DIR}/libsco_storage.a"
          "${SCO_CORE_HOST_BUILD_DIR}/libsco_host.a" "${SCO_CORE_HOST_BUILD_DIR}/libsco_runtime.a"
          "${SCO_CORE_HOST_BUILD_DIR}/libsco_core.a" "${SCO_CORE_HOST_BUILD_DIR}/libsco_sqlite.a" ${CMAKE_DL_LIBS})
      endif()
    endforeach()
    add_dependencies(gear-presets-workflow-test sc-offline-gear-presets)
    add_test(NAME gear-presets-storage COMMAND gear-presets-storage-test "${CMAKE_BINARY_DIR}/storage-fixture")
    add_test(NAME gear-presets-workflow COMMAND gear-presets-workflow-test "${CMAKE_BINARY_DIR}/workflow-fixture" $<TARGET_FILE:sc-offline-gear-presets>)
  endif()
endif()
