cmake_minimum_required(VERSION 3.20)
if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  message(FATAL_ERROR "Optional voxel bridge must be configured independently; never default release")
endif()
project(sc-offline-voxel-bridge LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(SCO_SDK_INCLUDE_DIR "" CACHE PATH "Checksum-verified extracted public SDK include directory")
if(NOT EXISTS "${SCO_SDK_INCLUDE_DIR}/../SHA256SUMS")
  message(FATAL_ERROR "Use a verified extracted SDK, not host headers")
endif()
add_library(sc-offline-voxel_bridge MODULE voxel_bridge.cpp)
target_include_directories(sc-offline-voxel_bridge PRIVATE "${SCO_SDK_INCLUDE_DIR}")
target_compile_features(sc-offline-voxel_bridge PRIVATE cxx_std_20)
set_target_properties(sc-offline-voxel_bridge PROPERTIES PREFIX "" OUTPUT_NAME "voxel_bridge" CXX_VISIBILITY_PRESET hidden)
if(MSVC)
  target_compile_options(sc-offline-voxel_bridge PRIVATE /W4 /WX /permissive-)
elseif(APPLE)
  target_compile_options(sc-offline-voxel_bridge PRIVATE -Wall -Wextra -Werror)
  target_link_options(sc-offline-voxel_bridge PRIVATE -Wl,-undefined,error)
else()
  target_compile_options(sc-offline-voxel_bridge PRIVATE -Wall -Wextra -Werror)
  target_link_options(sc-offline-voxel_bridge PRIVATE -Wl,--no-undefined)
endif()
install(TARGETS sc-offline-voxel_bridge LIBRARY DESTINATION data/plugins/voxel_bridge COMPONENT voxel_bridge RUNTIME DESTINATION data/plugins/voxel_bridge COMPONENT voxel_bridge)
install(FILES plugin.ini disabled DESTINATION data/plugins/voxel_bridge COMPONENT voxel_bridge)
option(SCO_WORKFLOW_TESTS "Build targeted synthetic module workflow" OFF)
if(SCO_WORKFLOW_TESTS)
  enable_testing()
  add_executable(voxel-bridge-module-test ../../tools/voxel_bridge_workflow_test.cpp)
  target_compile_features(voxel-bridge-module-test PRIVATE cxx_std_20)
  target_include_directories(voxel-bridge-module-test PRIVATE "${SCO_SDK_INCLUDE_DIR}")
  target_link_libraries(voxel-bridge-module-test PRIVATE ${CMAKE_DL_LIBS})
  if(MSVC)
    target_compile_options(voxel-bridge-module-test PRIVATE /UNDEBUG)
  else()
    target_compile_options(voxel-bridge-module-test PRIVATE -UNDEBUG)
  endif()
  add_dependencies(voxel-bridge-module-test sc-offline-voxel_bridge)
  add_test(NAME voxel-bridge-module-workflow COMMAND voxel-bridge-module-test $<TARGET_FILE:sc-offline-voxel_bridge>)
endif()
