cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(sc-offline-spawn LANGUAGES CXX)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(SCO_SDK_INCLUDE_DIR "" CACHE PATH "Verified published SDK include directory")
if(NOT EXISTS "${SCO_SDK_INCLUDE_DIR}/../SHA256SUMS")
  message(FATAL_ERROR "Use the extracted published SDK, not host headers")
endif()
foreach(header sco_widgets.h sco_api.h sco_content.h sc_spawn.h sc_entities.h sc_actors.h sc_vehicles.h sc_spatial.h)
  if(NOT EXISTS "${SCO_SDK_INCLUDE_DIR}/${header}")
    message(FATAL_ERROR "Missing published header: ${header}")
  endif()
endforeach()
add_library(sco-feature-spawn STATIC spawn.cpp)
target_include_directories(sco-feature-spawn PUBLIC "${SCO_SDK_INCLUDE_DIR}")
target_compile_features(sco-feature-spawn PUBLIC cxx_std_20)
set_target_properties(sco-feature-spawn PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden)
add_library(sc-offline-spawn MODULE external.cpp)
target_link_libraries(sc-offline-spawn PRIVATE sco-feature-spawn)
set_target_properties(sc-offline-spawn PROPERTIES PREFIX "" OUTPUT_NAME "spawn" CXX_VISIBILITY_PRESET hidden
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/test-plugins/spawn/$<0:>"
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/test-plugins/spawn/$<0:>")
if(APPLE)
  target_link_options(sc-offline-spawn PRIVATE -Wl,-undefined,error)
elseif(NOT WIN32)
  target_link_options(sc-offline-spawn PRIVATE -Wl,--no-undefined)
endif()
option(SCO_SPAWN_DEPENDENCY_TRACE "Emit MSVC compiler dependency inventory" OFF)
foreach(target sco-feature-spawn sc-offline-spawn)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
    if(SCO_SPAWN_DEPENDENCY_TRACE)
      set(trace_dir "${CMAKE_BINARY_DIR}/dependencies/${target}")
      file(MAKE_DIRECTORY "${trace_dir}")
      target_compile_options(${target} PRIVATE /sourceDependencies "${trace_dir}/")
      set(trace_props "${trace_dir}/includes.props")
      file(WRITE "${trace_props}" "<Project xmlns=\"http://schemas.microsoft.com/developer/msbuild/2003\"><Target Name=\"WorkflowToolchainIncludes\" BeforeTargets=\"ClCompile\"><WriteLinesToFile File=\"${trace_dir}/toolchain.txt\" Lines=\"$(IncludePath)\" Overwrite=\"true\" /></Target></Project>")
      set_target_properties(${target} PROPERTIES VS_USER_PROPS "${trace_props}")
    endif()
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  endif()
endforeach()
configure_file(plugin.ini "${CMAKE_BINARY_DIR}/test-plugins/spawn/plugin.ini" COPYONLY)
install(TARGETS sc-offline-spawn LIBRARY DESTINATION data/plugins/spawn COMPONENT spawn
  RUNTIME DESTINATION data/plugins/spawn COMPONENT spawn)
install(FILES plugin.ini DESTINATION data/plugins/spawn COMPONENT spawn)
option(SCO_WORKFLOW_TESTS "Build the synthetic dynamic module workflow" OFF)
if(SCO_WORKFLOW_TESTS)
  enable_testing()
  add_executable(spawn-module-test ../../tools/spawn_workflow_test.cpp)
  target_compile_features(spawn-module-test PRIVATE cxx_std_20)
  target_include_directories(spawn-module-test PRIVATE "${SCO_SDK_INCLUDE_DIR}")
  target_link_libraries(spawn-module-test PRIVATE ${CMAKE_DL_LIBS})
  add_dependencies(spawn-module-test sc-offline-spawn)
  add_test(NAME spawn-module-workflow COMMAND spawn-module-test $<TARGET_FILE:sc-offline-spawn>)
  if(EXISTS "${SCO_SDK_INCLUDE_DIR}/sc_asop_selection.h")
    add_executable(asop-selection-workflow-test ../../tools/asop_selection_workflow_test.cpp)
    target_compile_features(asop-selection-workflow-test PRIVATE cxx_std_20)
    target_include_directories(asop-selection-workflow-test PRIVATE "${SCO_SDK_INCLUDE_DIR}")
    target_link_libraries(asop-selection-workflow-test PRIVATE ${CMAKE_DL_LIBS})
    add_dependencies(asop-selection-workflow-test sc-offline-spawn)
    add_test(NAME asop-selection-workflow COMMAND asop-selection-workflow-test $<TARGET_FILE:sc-offline-spawn>)
  endif()
endif()
