# SDK-only feature targets. Configure this directory alone against an extracted core+game SDK.
cmake_minimum_required(VERSION 3.20)
if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  message(FATAL_ERROR "Configure plugins separately from the host, using a published SDK")
endif()
project(sc-offline-workflows LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(SCO_SDK_INCLUDE_DIR "" CACHE PATH "Public include directory of the core + game SDK")
set(SCO_SDK_VERSION "2.4.0" CACHE STRING "SDK archive version (published default: 2.4.0 RC2)")
set(SCO_SDK_SHA256 "51a7ce266d087bcbeac39c92f26a6f905b4473883dd7ed6f0f5ca823fa88c88e"
  CACHE STRING "SHA-256 of sco-sdk-<version>.zip")
set(SCO_SDK_ARCHIVE "" CACHE FILEPATH "Exact local SDK archive used for this configure (optional)")
option(SCO_INSPECTOR_DEPENDENCY_TRACE "Emit MSVC inspector compiler dependencies" OFF)
option(SCO_NPC_DEPENDENCY_TRACE "Emit MSVC NPC compiler dependencies" OFF)
if(NOT SCO_SDK_INCLUDE_DIR)
  message(FATAL_ERROR "Set SCO_SDK_INCLUDE_DIR to the installed SDK include directory")
endif()
if(NOT EXISTS "${SCO_SDK_INCLUDE_DIR}/../SHA256SUMS")
  message(FATAL_ERROR "Use an extracted published SDK with its checksum inventory, not host headers")
endif()
string(LENGTH "${SCO_SDK_SHA256}" _sco_sdk_hash_length)
if(NOT SCO_SDK_SHA256 MATCHES "^[0-9a-fA-F]+$" OR NOT _sco_sdk_hash_length EQUAL 64)
  message(FATAL_ERROR "SCO_SDK_SHA256 must be a 64-digit SHA-256")
endif()
if((NOT SCO_SDK_VERSION STREQUAL "2.4.0" OR
    NOT SCO_SDK_SHA256 STREQUAL "51a7ce266d087bcbeac39c92f26a6f905b4473883dd7ed6f0f5ca823fa88c88e")
   AND NOT SCO_SDK_ARCHIVE)
  message(FATAL_ERROR "Non-default SDK identity requires an explicit SCO_SDK_ARCHIVE to verify its identity")
endif()
if(SCO_SDK_ARCHIVE)
  if(NOT EXISTS "${SCO_SDK_ARCHIVE}")
    message(FATAL_ERROR "SCO_SDK_ARCHIVE does not exist: ${SCO_SDK_ARCHIVE}")
  endif()
  get_filename_component(_sco_sdk_archive_name "${SCO_SDK_ARCHIVE}" NAME)
  if(NOT _sco_sdk_archive_name STREQUAL "sco-sdk-${SCO_SDK_VERSION}.zip")
    message(FATAL_ERROR "SCO_SDK_ARCHIVE must be named sco-sdk-${SCO_SDK_VERSION}.zip")
  endif()
  file(SHA256 "${SCO_SDK_ARCHIVE}" _sco_sdk_archive_sha256)
  if(NOT _sco_sdk_archive_sha256 STREQUAL SCO_SDK_SHA256)
    message(FATAL_ERROR "SCO_SDK_ARCHIVE SHA-256 ${_sco_sdk_archive_sha256} does not match SCO_SDK_SHA256 ${SCO_SDK_SHA256}; refusing configure")
  endif()
endif()
foreach(header sco_api.h sc_actors.h sc_entities.h sc_spatial.h
               sc_actor_stats.h sc_game_build.h sc_gamerules.h sc_inventory.h sc_input.h sco_widgets.h)
  if(NOT EXISTS "${SCO_SDK_INCLUDE_DIR}/${header}")
    message(FATAL_ERROR "SDK public header missing: ${header}")
  endif()
endforeach()

foreach(feature npc inspector)
  add_library(sco-feature-${feature} STATIC ${feature}/${feature}.cpp)
  target_include_directories(sco-feature-${feature} PUBLIC "${SCO_SDK_INCLUDE_DIR}")
  target_compile_features(sco-feature-${feature} PUBLIC cxx_std_20)
  set_target_properties(sco-feature-${feature} PROPERTIES POSITION_INDEPENDENT_CODE ON
    CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
  add_library(sc-offline-${feature} MODULE ${feature}/external.cpp)
  target_link_libraries(sc-offline-${feature} PRIVATE sco-feature-${feature})
  if(APPLE)
    target_link_options(sc-offline-${feature} PRIVATE -Wl,-undefined,error)
  elseif(NOT WIN32)
    # ELF shared objects otherwise allow unresolved private host symbols.
    target_link_options(sc-offline-${feature} PRIVATE -Wl,--no-undefined)
  endif()
  set_target_properties(sc-offline-${feature} PROPERTIES PREFIX "" OUTPUT_NAME "${feature}"
    CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/test-plugins/${feature}/$<0:>"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/test-plugins/${feature}/$<0:>")
  foreach(target sco-feature-${feature} sc-offline-${feature})
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
      if(feature STREQUAL "inspector")
        target_compile_options(${target} PRIVATE /utf-8)
      endif()
      if((feature STREQUAL "inspector" AND SCO_INSPECTOR_DEPENDENCY_TRACE) OR
         (feature STREQUAL "npc" AND SCO_NPC_DEPENDENCY_TRACE))
        set(trace_dir "${CMAKE_BINARY_DIR}/dependencies/${target}")
        file(MAKE_DIRECTORY "${trace_dir}")
        target_compile_options(${target} PRIVATE /sourceDependencies "${trace_dir}/")
        # VS supplies CRT/Windows SDK includes outside CMake's implicit paths.
        # Capture the evaluated IncludePath at the actual compile target.
        set(trace_props "${trace_dir}/includes.props")
        file(WRITE "${trace_props}"
          "<Project xmlns=\"http://schemas.microsoft.com/developer/msbuild/2003\"><Target Name=\"WorkflowToolchainIncludes\" BeforeTargets=\"ClCompile\"><WriteLinesToFile File=\"${trace_dir}/toolchain.txt\" Lines=\"$(IncludePath)\" Overwrite=\"true\" /></Target></Project>")
        set_target_properties(${target} PROPERTIES VS_USER_PROPS "${trace_props}")
      endif()
    else()
      target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    endif()
  endforeach()
  configure_file(${feature}/plugin.ini "${CMAKE_BINARY_DIR}/test-plugins/${feature}/plugin.ini" COPYONLY)
  # A targeted proof builds only one module; install only that feature component.
  install(TARGETS sc-offline-${feature} LIBRARY DESTINATION data/plugins/${feature}
    COMPONENT ${feature} RUNTIME DESTINATION data/plugins/${feature} COMPONENT ${feature})
  install(FILES ${feature}/plugin.ini DESTINATION data/plugins/${feature} COMPONENT ${feature})
endforeach()
