# SDK-only feature targets. Configure this directory alone against an extracted core+game SDK.
cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(sc-offline-workflows LANGUAGES CXX)
endif()
set(SCO_SDK_INCLUDE_DIR "" CACHE PATH "Public include directory of the core + game SDK")
if(NOT SCO_SDK_INCLUDE_DIR)
  message(FATAL_ERROR "Set SCO_SDK_INCLUDE_DIR to the installed SDK include directory")
endif()
foreach(header sco_api.h sc_actors.h sc_entities.h sc_spatial.h)
  if(NOT EXISTS "${SCO_SDK_INCLUDE_DIR}/${header}")
    message(FATAL_ERROR "SDK public header missing: ${header}")
  endif()
endforeach()

foreach(feature npc inspector)
  add_library(sco-feature-${feature} STATIC ${feature}/${feature}.cpp)
  target_include_directories(sco-feature-${feature} PUBLIC "${SCO_SDK_INCLUDE_DIR}")
  target_compile_features(sco-feature-${feature} PUBLIC cxx_std_20)
  set_target_properties(sco-feature-${feature} PROPERTIES POSITION_INDEPENDENT_CODE ON
    CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
  add_library(sc-offline-${feature} MODULE ${feature}/external.cpp)
  target_link_libraries(sc-offline-${feature} PRIVATE sco-feature-${feature})
  if(APPLE)
    target_link_options(sc-offline-${feature} PRIVATE -Wl,-undefined,error)
  elseif(NOT WIN32)
    # ELF shared objects otherwise allow unresolved private host symbols.
    target_link_options(sc-offline-${feature} PRIVATE -Wl,--no-undefined)
  endif()
  set_target_properties(sc-offline-${feature} PROPERTIES PREFIX "" OUTPUT_NAME "${feature}"
    CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/test-plugins/${feature}/$<0:>"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/test-plugins/${feature}/$<0:>")
  foreach(target sco-feature-${feature} sc-offline-${feature})
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
    else()
      target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    endif()
  endforeach()
  configure_file(${feature}/plugin.ini "${CMAKE_BINARY_DIR}/test-plugins/${feature}/plugin.ini" COPYONLY)
endforeach()

option(SCO_WORKFLOW_TESTS "Build the targeted synthetic workflow tests" OFF)
if(SCO_WORKFLOW_TESTS)
  enable_testing()
  add_executable(npc-workflow-test ../tools/npc_workflow_test.cpp npc/external.cpp)
  target_compile_definitions(npc-workflow-test PRIVATE NPC_EXTERNAL_TEST)
  target_link_libraries(npc-workflow-test PRIVATE sco-feature-npc)
  add_test(NAME npc-workflow COMMAND npc-workflow-test)
  add_executable(inspector-test ../tools/inspector_test.cpp inspector/external.cpp)
  target_compile_definitions(inspector-test PRIVATE INSPECTOR_EXTERNAL_TEST)
  target_link_libraries(inspector-test PRIVATE sco-feature-inspector)
  add_test(NAME inspector-workflow COMMAND inspector-test)
  foreach(feature npc inspector)
    if(feature STREQUAL "npc")
      set(test_source ../tools/npc_workflow_test.cpp)
    else()
      set(test_source ../tools/inspector_test.cpp)
    endif()
    add_executable(${feature}-module-test ${test_source})
    target_compile_features(${feature}-module-test PRIVATE cxx_std_20)
    target_compile_definitions(${feature}-module-test PRIVATE WORKFLOW_DYNAMIC_TEST)
    target_include_directories(${feature}-module-test PRIVATE "${SCO_SDK_INCLUDE_DIR}")
    target_link_libraries(${feature}-module-test PRIVATE ${CMAKE_DL_LIBS})
    add_dependencies(${feature}-module-test sc-offline-${feature})
    add_test(NAME ${feature}-module-workflow COMMAND ${feature}-module-test $<TARGET_FILE:sc-offline-${feature}>)
  endforeach()
  # The registration adapter intentionally uses the host kit, not the installed public SDK.
  set(SCO_HOST_INCLUDE_DIR "" CACHE PATH "Optional pinned host headers for built-in adapter tests")
  if(SCO_HOST_INCLUDE_DIR)
    foreach(feature npc inspector)
      if(feature STREQUAL "npc")
        set(test_source ../tools/npc_workflow_test.cpp)
      else()
        set(test_source ../tools/inspector_test.cpp)
      endif()
      add_executable(${feature}-builtin-test ${test_source} ../src/workflow_adapters.cpp)
      string(TOUPPER "${feature}" feature_upper)
      target_compile_definitions(${feature}-builtin-test PRIVATE ${feature_upper}_BUILTIN_TEST)
      target_include_directories(${feature}-builtin-test PRIVATE "${SCO_HOST_INCLUDE_DIR}")
      target_link_libraries(${feature}-builtin-test PRIVATE sco-feature-npc sco-feature-inspector)
      add_test(NAME ${feature}-builtin-workflow COMMAND ${feature}-builtin-test)
    endforeach()
  endif()
endif()
