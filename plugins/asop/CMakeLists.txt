cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(sc-offline-asop-selector LANGUAGES CXX)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(SCO_SDK_INCLUDE_DIR "" CACHE PATH "Verified published SDK include directory")
if(NOT EXISTS "${SCO_SDK_INCLUDE_DIR}/../SHA256SUMS")
  message(FATAL_ERROR "Use a verified published SDK archive")
endif()
add_library(sc-offline-asop-selector MODULE asop.cpp)
target_compile_features(sc-offline-asop-selector PRIVATE cxx_std_20)
target_include_directories(sc-offline-asop-selector PRIVATE "${SCO_SDK_INCLUDE_DIR}")
set_target_properties(sc-offline-asop-selector PROPERTIES PREFIX "" OUTPUT_NAME "asop_selector"
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/test-plugins/asop_selector/$<0:>"
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/test-plugins/asop_selector/$<0:>")
if(MSVC)
  target_compile_options(sc-offline-asop-selector PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(sc-offline-asop-selector PRIVATE -Wall -Wextra -Werror)
endif()
configure_file(plugin.ini "${CMAKE_BINARY_DIR}/test-plugins/asop_selector/plugin.ini" COPYONLY)
install(TARGETS sc-offline-asop-selector LIBRARY DESTINATION data/plugins/asop_selector COMPONENT asop_selector RUNTIME DESTINATION data/plugins/asop_selector COMPONENT asop_selector)
install(FILES plugin.ini DESTINATION data/plugins/asop_selector COMPONENT asop_selector)
option(SCO_WORKFLOW_TESTS "Build ASOP selector dynamic fixture" OFF)
if(SCO_WORKFLOW_TESTS)
  enable_testing()
  add_executable(asop-selector-test ../../tools/asop_workflow_test.cpp)
  target_compile_features(asop-selector-test PRIVATE cxx_std_20)
  target_include_directories(asop-selector-test PRIVATE "${SCO_SDK_INCLUDE_DIR}")
  if(MSVC)
    target_compile_options(asop-selector-test PRIVATE /UNDEBUG /W4 /WX)
  endif()
  target_link_libraries(asop-selector-test PRIVATE ${CMAKE_DL_LIBS})
  add_test(NAME asop-selector-workflow COMMAND asop-selector-test $<TARGET_FILE:sc-offline-asop-selector>)
endif()
