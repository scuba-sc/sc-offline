cmake_minimum_required(VERSION 3.20)
if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  message(FATAL_ERROR "Configure TitanLink independently against a verified public SDK")
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(sc-offline-titanlink LANGUAGES CXX)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(SCO_SDK_INCLUDE_DIR "" CACHE PATH "Verified published SDK include directory")
if(NOT EXISTS "${SCO_SDK_INCLUDE_DIR}/../SHA256SUMS")
  message(FATAL_ERROR "Use the extracted published SDK, not host headers")
endif()
foreach(header sco_api.h sc_titanlink.h sc_entities.h sc_actors.h sc_vehicles.h sc_spatial.h)
  if(NOT EXISTS "${SCO_SDK_INCLUDE_DIR}/${header}")
    message(FATAL_ERROR "Missing published header: ${header}")
  endif()
endforeach()
add_library(sco-feature-titanlink STATIC titanlink.cpp platform.cpp overlay.cpp)
target_include_directories(sco-feature-titanlink PUBLIC "${SCO_SDK_INCLUDE_DIR}")
target_compile_features(sco-feature-titanlink PUBLIC cxx_std_20)
set_target_properties(sco-feature-titanlink PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden)
add_library(sc-offline-titanlink MODULE external.cpp)
target_link_libraries(sc-offline-titanlink PRIVATE sco-feature-titanlink)
set_target_properties(sc-offline-titanlink PROPERTIES PREFIX "" OUTPUT_NAME "titanlink" CXX_VISIBILITY_PRESET hidden
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/test-plugins/titanlink/$<0:>"
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/test-plugins/titanlink/$<0:>")
if(APPLE)
  target_link_options(sc-offline-titanlink PRIVATE -Wl,-undefined,error)
elseif(NOT WIN32)
  target_link_options(sc-offline-titanlink PRIVATE -Wl,--no-undefined)
endif()
option(SCO_TITANLINK_DEPENDENCY_TRACE "Emit MSVC compiler dependency inventory" OFF)
foreach(target sco-feature-titanlink sc-offline-titanlink)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
    if(SCO_TITANLINK_DEPENDENCY_TRACE)
      set(trace_dir "${CMAKE_BINARY_DIR}/dependencies/${target}")
      file(MAKE_DIRECTORY "${trace_dir}")
      target_compile_options(${target} PRIVATE /sourceDependencies "${trace_dir}/")
      set(trace_props "${trace_dir}/includes.props")
      file(WRITE "${trace_props}" "<Project xmlns=\"http://schemas.microsoft.com/developer/msbuild/2003\"><Target Name=\"WorkflowToolchainIncludes\" BeforeTargets=\"ClCompile\"><WriteLinesToFile File=\"${trace_dir}/toolchain.txt\" Lines=\"$(IncludePath)\" Overwrite=\"true\" /></Target></Project>")
      set_target_properties(${target} PROPERTIES VS_USER_PROPS "${trace_props}")
    endif()
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  endif()
endforeach()
configure_file(plugin.ini "${CMAKE_BINARY_DIR}/test-plugins/titanlink/plugin.ini" COPYONLY)
install(TARGETS sc-offline-titanlink LIBRARY DESTINATION data/plugins/titanlink COMPONENT titanlink
  RUNTIME DESTINATION data/plugins/titanlink COMPONENT titanlink)
install(FILES disabled plugin.ini DESTINATION data/plugins/titanlink COMPONENT titanlink)
option(SCO_WORKFLOW_TESTS "Build the synthetic dynamic module workflow" OFF)
if(SCO_WORKFLOW_TESTS)
  enable_testing()
  add_executable(titanlink-module-test ../../tools/titanlink_workflow_test.cpp)
  target_compile_features(titanlink-module-test PRIVATE cxx_std_20)
  target_include_directories(titanlink-module-test PRIVATE "${SCO_SDK_INCLUDE_DIR}")
  target_link_libraries(titanlink-module-test PRIVATE ${CMAKE_DL_LIBS})
  add_dependencies(titanlink-module-test sc-offline-titanlink)
  add_test(NAME titanlink-module-workflow COMMAND titanlink-module-test $<TARGET_FILE:sc-offline-titanlink>)
  set_tests_properties(titanlink-module-workflow PROPERTIES TIMEOUT 30)
endif()

if(WIN32)
  target_link_libraries(sc-offline-titanlink PRIVATE d3d11 d3dcompiler dcomp dxgi advapi32 user32)
endif()
